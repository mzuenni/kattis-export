\begin{slide}
	\begin{block}{Problem}
		Given the $n$ song lengths of a playlist (in \texttt{m:ss} format) and a crossfade of $c$ seconds, how long does it take to listen to the entire playlist (in \texttt{hh:mm:ss} format)?
	\end{block}
	\pause
	\begin{block}{Solution}
		\begin{itemize}
			\item Convert \texttt{m:ss} input lengths into (only) seconds.
			\item Subtract $(n-1) \cdot c$ seconds.
			\item Convert to \texttt{hh:mm:ss} format.
			\item Careful: \texttt{1:00} minus $10$ seconds should not become \texttt{1:-10}!
		\end{itemize}
	\end{block}
\end{slide}
