\begin{slide}
	\begin{block}{Problem}
		Two players are on a grid, and each has to move a certain number of times. Each move is to an adjacent cell (no diagonals, no staying in place). Determine a cell they can both end up on after their move, or state that there is none.
	\end{block}
	\pause
	\begin{block}{Solution}
		\begin{itemize}
			\item If $abs(x_1 - x_2) + abs(y_1 - y_2) > d_1 + d_2$, the players are too far away from each other -> \texttt{impossible}.
			\item Each move changes the player's $x$ or $y$ coordinate by $1$. Therefore, $x_1 + y_1 + d_1 \equiv x_2 + y_2 + d_2 \ mod \ 2$ must be satisfied, otherwise \texttt{impossible}.
			\item No other condition leads to an \texttt{impossible}.
		\end{itemize}
	\end{block}
\end{slide} 
\begin{slide}
	\begin{block}{Finding a valid target cell}
		Consider the smaller of the two ranges. The corner closest to the other player's starting point is a valid cell.
	\end{block}
