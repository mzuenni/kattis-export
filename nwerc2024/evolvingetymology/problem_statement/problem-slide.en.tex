        \begin{column}[T]{.62\textwidth}
            \begin{itemize}
                \item Create a new word from an existing word $s$ ($|s| \leq 10^5$)
                by applying the following method up to $10^{18}$ times:
                \begin{itemize}
                    \normalsize
                    \item Take every second letter of $s+s$, starting with the first letter.
                \end{itemize}
                \item For example, applying this method to the word ``\texttt{etymology}'' once
                    would result in ``\texttt{eyooytmlg}''.
            \end{itemize}
