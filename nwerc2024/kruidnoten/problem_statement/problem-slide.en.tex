        \begin{column}[T]{0.67\textwidth}
            \begin{itemize}
                \item You are a given an undirected, weighted graph of $n \le 2\cdot10^5$ vertices and $m \le 2\cdot10^5$ edges.
                \item $k \le n$ of the vertices contain a store that \textit{may} sell kruidnoten.
                \item Store $i$ sells kruidnoten with probability $0 < p_i \le 1$.
                \item Karlijn takes a path from vertex $1$ to vertex $n$ that must contain at least one store that sells kruidnoten.
                \item Find the expected length of a shortest path from $1$ to $n$.
            \end{itemize}
        \end{column}
