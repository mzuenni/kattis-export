        \begin{column}[T]{.72\textwidth}
            \begin{itemize}
                \item You have a bookcase with $n \leq 10^5$ shelves with given heights up~to~$10^9$.
                \begin{itemize}
                    \normalsize
		        \item Each shelf fits at most $x \leq 1000$ books on a full shelf (without~art~piece).
		        \item Each shelf fits at most $y < x$ books on a shelf with art piece.
                \end{itemize}
                \item You have $m \leq 10^5$ books with given heights up to $10^9$.
                \begin{itemize}
                    \normalsize
                    \item All books have the same width.
                \end{itemize}
                \item If possible, you should fit all books on the shelf, and output the largest number of art pieces you can place. Else, output ``impossible''.
            \end{itemize}
