\begin{frame}
    \frametitle{\problemtitle}
    \begin{block}{Problem}
        Given a list of $n\leq1000$ allegedly Dutch surnames, sort them according to the part starting from the first capital letter.
    \end{block}
    \pause
    \begin{block}{Solution}
        \begin{itemize}
            \item Use a sort function from the standard library of your favourite programming language.
            \item Specify a custom comparator or sorting key.
            \item Default string comparisons are already lexicographic and compare using ASCII values.
        \end{itemize}
        Run time: $\mathcal O(n\log n)$.
    \end{block}
    \pause
    \solvestats
\end{frame}
