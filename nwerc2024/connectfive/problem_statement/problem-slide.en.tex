        \begin{column}[T]{\textwidth}
            \begin{itemize}
                \item Given are five locations in a city grid with square blocks, with coordinates up to $1000$.
                \item Refurbish a set of streets in the grid, such that:
                \begin{itemize}
                    \item for any two locations there exists a shortest path between them that only uses refurbished streets; and
                    \item the total length of the refurbished streets is minimized.
                \end{itemize}
                \item Output the minimum number of road segments that need to be refurbished.
            \end{itemize}
