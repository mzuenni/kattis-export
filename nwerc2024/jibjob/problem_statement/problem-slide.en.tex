        \begin{column}[T]{.62\textwidth}
            \begin{itemize}
                \item Given are $n \leq 500$ cranes, with coordinates and heights up to $10\,000$.
                \item For each crane, output the positive integer length of its jib in metres, such that the covered area is maximized.
                \item If there are multiple optimal solutions, you may output any one of them.
            \end{itemize}
        \end{column}
