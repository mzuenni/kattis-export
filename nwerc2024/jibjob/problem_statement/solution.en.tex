\begin{frame}
    \frametitle{\problemtitle}
    \begin{block}{Problem}
    	Given $n$ crane towers with different positions and heights.
    	Place a \emph{jib} on top of each crane such that:
    	\begin{itemize}
    		\item No jib can collide with any other crane tower.
    		\item No crane is unstable by having a jib longer than the tower's height.
    		\item The ground area that can be reached is maximized.
    	\end{itemize}
    \end{block}
    \bigskip
    \centering

\begin{frame}
	\frametitle{\problemtitle}
	\begin{block}{Observation}
		\begin{itemize}
			\item The size of a jib does not influence any other jib since only jibs and cranes can collide.
			\item Larger jibs cannot cover less area.
			\pause
			\item[] $\Longrightarrow$ Make each jib as large as possible.
		\end{itemize}
	\end{block}
	\bigskip
	\pause
	\begin{block}{Solution}
		For each tower, determine the closest tower that is larger. Take the minimum of that distance and the height of the tower.
		\pause

		Runtime: $\mathcal{O}(n^2)$
	\end{block}
	\pause
	\solvestats
