        \begin{column}[T]{\textwidth}
            \begin{itemize}
                \item There are $n\leq 60$ parties with $p\leq 10\,000$ seats each.
                \item Count the number of candidate coalitions meeting these two criteria:
                \begin{description}
                    \item[Strict Majority:] The total number of seats held by the coalition must be strictly more than half of the total seats across all parties.
                    \item[No Superfluous Parties:] The coalition must be minimal in the sense that removing any one party from the coalition would cause it to lose its strict majority.
                \end{description}
            \end{itemize}
