\begin{slide}
    \begin{block}{Problem}
        Find the maximum sum of difficulty rating of practice contests that Gregor can actually take part in.
    \end{block}
\end{slide}

\begin{slide}

\begin{itemize}
  \item Weighted Interval Scheduling Problem
  \item Sort intervals by finishing time.
  \item Use dynamic programming over the intervals (1-based).
  \item Let $p(i)$ be the index of the latest finishing interval that is still compatible with interval $i$.
\end{itemize}
\begin{align*}
  \texttt{dp}[0] &= 0\\
  \texttt{dp}[i] &= \max(\texttt{dp}[i-1], \texttt{dp}[p(i)] + weight(i))
\end{align*}
\begin{itemize}
  \item Compute $\texttt{dp}[n]$.
\end{itemize}

\end{slide}
