        \begin{column}[T]{.70\textwidth}
            \begin{itemize}
                \item Given are $n \leq 100$ top-10 lists.
                \item Find the artist with the highest number of appearances.
                \item In case of a tie, look at most 1st-place positions.
                \item When still a tie, look at most 2nd-place positions.
                \item And so on\ldots
                \item If they are tied for all $10$ positions, output ``\texttt{tie}''.
            \end{itemize}
