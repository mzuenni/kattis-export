    \begin{columns}
        \begin{column}[T]{.70\textwidth}
            \begin{itemize}
                \item Create an $h\times w$ ($h,w \le 100$) grid consisting of letters `\texttt{K}', `\texttt{I}' and `\texttt{T}' such that
                \begin{itemize}
                    \normalsize
                    \item each letter occurs a given number of times ($k$, $i$ and $t$ times, $k+i+t = h\cdot w$); and
                    \item there is exactly one occurrence of the word ``\texttt{KIT}''.
                \end{itemize}
                \item Words can be read in $8$ directions: horizontally, vertically, diagonally.
            \end{itemize}
