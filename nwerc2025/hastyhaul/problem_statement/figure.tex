\newcommand{\twobytwo}[7]{
   \matrix (#1) [matrix of nodes,
                 nodes={minimum size=5mm, anchor=center, inner sep=0pt,draw},
                 row sep=-\pgflinewidth,
                 column sep=-\pgflinewidth,
                 ampersand replacement=\&
                 ] at (#2, #3)
   {
     #4 \& #5 \\ 
     #6 \& #7 \\
   };
}

\newcommand{\onebythree}[6]{

                 row sep=-\pgflinewidth,
                 column sep=-\pgflinewidth,
                 ampersand replacement=\&
                 ] at (#2, #3)
   {
     #4 \& #5 \& #6 \\ 
   };
}
