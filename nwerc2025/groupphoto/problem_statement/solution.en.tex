\begin{frame}
    \frametitle{\problemtitle}
    \begin{block}{Problem}
		Given a permutation, reorder as few elements as possible to make it
		mountain-shaped (first increasing then decreasing) in $\mathcal{O}(n \log n)$.
    \end{block}
    \pause
    \begin{block}{First insight}
		If we place elements from smallest to largest, each element will go to
		one of the borders.

		This leads to a quadratic DP with state space
		\[(\text{Num elements placed on the left so far, Num elements placed on the right so far}).\]

		Transition value is $0$ if element is misplaced, and $1$ if it stays where it is.
    \end{block}
\end{frame}
