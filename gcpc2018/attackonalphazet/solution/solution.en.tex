\begin{slide}
  \begin{block}{Problem}
  Given a sequence of locations inside a maze, what is the minimal distance one has
  to travel to visit them all in given order?
  \end{block}
  \pause
  \begin{block}{Observation}
    \begin{itemize}
      \item The number of locations is too big for multiple iterations of any search algorithm.
      \item Since the maze has no loops, it can be seen as a tree.
    \end{itemize}
  \end{block}
\end{slide}

  \begin{block}{Solution}
    \begin{itemize}
      \item Transform the maze into a tree e.g. with \emph{depth first search}.
      \item Find \emph{Lowest Common Ancestors} to calculate the distance between 
        two consecutive locations.
      \item $\Rightarrow$ Init: $\mathcal{O}(n \log n)$, Lookup: $\mathcal{O}(1)$
    \end{itemize}
  \end{block}
