\begin{slide}
	\begin{block}{Problem}
		Given a non-negative integer sequence $A$, find the number of possible non-negative integer sequences $B$ such that $A_i = B_i + B_{i+1}$ for all pairs of adjacent numbers in $B$.
	\end{block}
	\pause
	\begin{block}{Insights}
		\begin{itemize}
			\item $B_1$ uniquely determines the rest of the sequence.
      \item If $B_1$ increases, each odd-numbered element increases
        and each even-numbered element decreases.
    \end{itemize}
	\end{block}
	\pause
  \begin{block}{Solution}
		\begin{itemize}
			\item Put $B_1 := 0$ and compute the resulting sequence.
      \item Find the minimum odd- and even-numbered elements.
			\item The range of answers is $[-\min\{B_{2i+1}\}, \min\{B_{2i}\}]$.
      \item Time complexity: $\mathcal{O}(n)$.
		\end{itemize}
	\end{block}
\end{slide}
