    \begin{block}{Problem}
      Given $n$ cables with lengths $d_1,\dots,d_n$, find the set of cables that has the \emph{largest minimal overlap} if fitted in the gap $g$.
    \end{block}
    \pause
    \begin{block}{Solution}
        \begin{itemize}
          \item Calculate all pairs $(i,j)$ such that there is a set of $i$ cables with total length $j$.
          \item This can be done with a knapsack-like DP in $\mathcal{O}(n^2 \cdot g)$.
          \item The largest minimal overlap for a pair $(i,j)$ is given by $\frac{j-g}{i+1}.$
          \item Try all pairs to find the best solution.
        \end{itemize}
    \end{block}
\end{slide}
