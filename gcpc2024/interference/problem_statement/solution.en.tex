\begin{frame}
    \frametitle{\problemtitle}
    \begin{block}{Problem}
    \begin{itemize}
        \item Given some alternating range updates, answer point queries
    \end{itemize}
    \end{block}
    \pause
    \begin{block}{Solution}
    \begin{itemize}
        \item<+-> Range is too large to work with, ignore it
        \item<+-> $\mathcal O(n^2)$ is sufficient, for each query simulate all former updates.
        \item<+-> Pitfalls: Results can be large (long long may be needed) or negative, correctly deal with alternation
    \end{itemize}
    \end{block}
    % \solvestats
\end{frame}
